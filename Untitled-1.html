Installation

Two lines. Any HTML page. Instant transformation.

<!-- Add in <head> --> <link rel="stylesheet" href="divine-horror-theme.css"> <!-- Add before </body> --> <script src="divine-horror-theme.js"></script> <!-- Optional: JS API --> <script> DivinHorror.setMode('devil'); DivinHorror.toggle(); DivinHorror.skeleton(el, true); DivinHorror.addGlitch(el); </script>
<!-- Skeleton on any element --> <div class="dh-skeleton">...</div> <!-- Glitch text (devil mode) --> <h1 class="dh-glitch" data-text="YOUR TEXT"> YOUR TEXT </h1> <!-- Sacred gradient text --> <h2 class="dh-sacred-text">Title</h2> <!-- Glow card --> <div class="dh-card">...</div>
